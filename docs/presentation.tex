% ============== SLIDE 3: CONCEPTE RL ==============
\begin{frame}{Concepte fundamentale RL}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Markov Decision Process (MDP)}
        \begin{itemize}
          \item $\mathcal{S}$ — spațiul stărilor
          \item $\mathcal{A}$ — spațiul acțiunilor
          \item $P(s'|s,a)$ — funcția de tranziție
          \item $R(s,a,s')$ — funcția de recompensă
          \item $\gamma \in [0,1]$ — factor de discount
        \end{itemize}
      \end{block}
      
      \begin{block}{Obiectiv}
        Maximizarea recompensei cumulate:
        \[G_t = \sum_{k=0}^{\infty} \gamma^k r_{t+k+1}\]
      \end{block}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{block}{Funcții cheie}
        \textbf{Value Function:}
        \[V^\pi(s) = \mathbb{E}_\pi[G_t | s_t = s]\]
        
        \textbf{Q-Function (Action-Value):}
        \[Q^\pi(s,a) = \mathbb{E}_\pi[G_t | s_t = s, a_t = a]\]
        
        \textbf{Policy:}
        \[\pi(a|s) = P(a_t = a | s_t = s)\]
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ============== SLIDE 4: BELLMAN ==============
\begin{frame}{Ecuația Bellman — baza algoritmilor}
  \begin{block}{Ecuația Bellman pentru Q-values}
    \[Q(s,a) = \mathbb{E}\left[r + \gamma \max_{a'} Q(s', a')\right]\]
  \end{block}
  
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Update iterativ (Q-Learning):}
      \[Q(s,a) \leftarrow Q(s,a) + \alpha \cdot \underbrace{(r + \gamma \max_{a'} Q(s',a') - Q(s,a))}_{\text{TD error}}\]
      
      \vspace{0.2cm}
      \textbf{Parametri:}
      \begin{itemize}
        \item $\alpha$ — learning rate (cât de mult actualizăm)
        \item $\gamma$ — discount (importanța recompenselor viitoare)
        \item TD error — diferența dintre target și estimare
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}[>=Stealth, scale=0.8]
        \node[circle, draw, fill=blue!20] (s) at (0,0) {$s$};
        \node[circle, draw, fill=green!20] (s1) at (2,1) {$s'$};
        \node[circle, draw, fill=green!20] (s2) at (2,-1) {$s''$};
        \draw[->, thick] (s) -- node[above]{$a, r_1$} (s1);

      \end{tikzpicture}
      
      {\footnotesize Backup diagram}
    \end{column}
  \end{columns}
\end{frame}

% ============== SLIDE 9: CODE Q-LEARNING ==============
\begin{frame}[fragile]{Cod: Tabular Q-Learning}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{lstlisting}[style=pythonstyle]
class TabularQAgent:
    def __init__(self, obs_bins, action_size, 
                 alpha=0.05, gamma=0.99, epsilon=1.0):
        self.q_table = np.zeros(obs_bins + [action_size])
        self.alpha = alpha    # learning rate
        self.gamma = gamma    # discount factor
        self.epsilon = epsilon

    def select_action(self, state):
        """Epsilon-greedy action selection."""
        if np.random.rand() < self.epsilon:
            return np.random.randint(self.action_size)
        return np.argmax(self.q_table[state])

    def update(self, state, action, reward, next_state, done):
        """Bellman update."""
        best_next = np.max(self.q_table[next_state])
        target = reward + self.gamma * best_next * (1 - done)
        error = target - self.q_table[state + (action,)]
        self.q_table[state + (action,)] += self.alpha * error
      \end{lstlisting}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Explicație:}
      \begin{itemize}
        \item \texttt{q\_table}: matrice N-dim pentru Q-values
        \item \texttt{select\_action}: explorare vs exploatare
        \item \texttt{update}: formula Bellman
        \item TD error = target $-$ Q curent
      \end{itemize}
      
      \vspace{0.3cm}
      \textbf{Discretizare:}
      
      Stare continuă $\to$ index în tabel
      
      $[x, y, fuel, ...] \to (2, 3, 1, ...)$
    \end{column}
  \end{columns}
\end{frame}

class DQNNetwork(nn.Module):
    def __init__(self, obs_dim, action_size, 
                 hidden_dims=[256, 128]):
        super().__init__()
        layers = []
        input_dim = obs_dim
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            input_dim = hidden_dim
        self.hidden = nn.Sequential(*layers)
        self.output = nn.Linear(hidden_dims[-1], action_size)

    def forward(self, state):
        x = self.hidden(state)
        return self.output(x)  # Q-values
      \end{lstlisting}

    \begin{column}{0.45\textwidth}
      \textbf{Arhitectura:}
      \begin{itemize}
        \item Input: 25 features (stare)
        \item Hidden: 256 $\to$ 128 neuroni

    \begin{column}{0.42\textwidth}
      \textbf{Policy Gradient:}
      \[\nabla J \approx \sum_t \nabla \log \pi(a_t|s_t) \cdot A_t\]
      
      \vspace{0.2cm}
      \textbf{Componente:}
      \begin{itemize}
        \item \textbf{Policy Net}: $\pi(a|s)$ — probabilități acțiuni
        \item \textbf{Value Net}: $V(s)$ — baseline pentru variance reduction
        \item \textbf{Advantage}: $A_t = G_t - V(s_t)$
      \end{itemize}
      
      \vspace{0.2cm}
      \textbf{Monte Carlo:}
      
      Așteptăm sfârșitul episodului pentru $G_t$
    \end{column}
